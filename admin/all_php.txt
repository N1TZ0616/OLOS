// Manage session-based cart (add, remove, update_all, clear)

if (session_status() === PHP_SESSION_NONE) session_start();
require_once __DIR__ . '/db_connect.php';

$action = $_REQUEST['action'] ?? '';
$method = $_SERVER['REQUEST_METHOD'];

if (!isset($_SESSION['cart']) || !is_array($_SESSION['cart'])) $_SESSION['cart'] = [];
$cart = &$_SESSION['cart'];

function getProduct($mysqli, $id) {
    $stmt = $mysqli->prepare("SELECT `id`,`name`,`price`,`image` FROM `menu` WHERE `id` = ? LIMIT 1");
    if (!$stmt) return null;
    $stmt->bind_param('i', $id);
    if ($stmt->execute()) {
        $res = $stmt->get_result();
        $row = $res->fetch_assoc();
        $res->free();
        $stmt->close();
        return $row;
    }
    $stmt->close();
    return null;
}

if ($action === 'add' && $method === 'POST') {
    $pid = isset($_POST['product_id']) ? (int)$_POST['product_id'] : 0;
    $qty = max(1, (int)($_POST['qty'] ?? 1));
    if ($pid > 0) {
        $prod = getProduct($mysqli, $pid);
        if ($prod) {
            if (isset($cart[$pid])) {
                $cart[$pid]['qty'] += $qty;
            } else {
                $cart[$pid] = [
                    'id' => $prod['id'],
                    'name' => $prod['name'],
                    'price' => (float)$prod['price'],
                    'image' => $prod['image'],
                    'qty' => $qty
                ];
            }
        }
    }
    $redirect = $_SERVER['HTTP_REFERER'] ?? 'menu.php';
    header('Location: ' . $redirect);
    exit;
}

if ($action === 'remove') {
    $pid = isset($_GET['product_id']) ? (int)$_GET['product_id'] : 0;
    if ($pid > 0 && isset($cart[$pid])) unset($cart[$pid]);
    header('Location: cart.php'); exit;
}

if ($action === 'update_all' && $method === 'POST') {
    $qtys = $_POST['qty'] ?? [];
    foreach ($qtys as $pid => $q) {
        $pid = (int)$pid; $q = (int)$q;
        if ($q < 1) {
            unset($cart[$pid]);
        } else {
            if (isset($cart[$pid])) $cart[$pid]['qty'] = $q;
        }
    }
    header('Location: cart.php'); exit;
}

if ($action === 'clear') {
    $_SESSION['cart'] = [];
    header('Location: cart.php'); exit;
}

header('Location: menu.php');
exit;

    $email = trim($_POST['email'] ?? '');
    $password = $_POST['password'] ?? '';
    $confirm = $_POST['confirm'] ?? '';

    if ($username === '') $errors[] = 'Username is required.';
    if ($email === '' || !filter_var($email, FILTER_VALIDATE_EMAIL)) $errors[] = 'Valid email is required.';
    if ($password === '' || strlen($password) < 6) $errors[] = 'Password must be at least 6 characters.';
    if ($password !== $confirm) $errors[] = 'Password confirmation does not match.';

    if (empty($errors)) {
        $hash = password_hash($password, PASSWORD_DEFAULT);
        $stmt = $mysqli->prepare("INSERT INTO `users` (`username`,`password_hash`,`full_name`,`email`,`role`) VALUES (?,?,?,?,?)");
        if ($stmt) {
            $role = 'customer';
            $stmt->bind_param('sssss', $username, $hash, $full_name, $email, $role);
            if ($stmt->execute()) {
                $success = 'Registration successful. You may login now.';
                $username = $full_name = $email = '';
            } else {
                $errors[] = 'Registration failed: ' . $stmt->error;
            }
            $stmt->close();
        } else {
            $errors[] = 'Service temporarily unavailable.';
            error_log('Register prepare error: ' . $mysqli->error);
        }
    }
}
